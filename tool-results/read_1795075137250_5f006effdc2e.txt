   135→const faNum = (n: number) => new Intl.NumberFormat("fa-IR").format(n);

   137→const toFaDigits = (s: string) =>

   140→function parseMoney(raw: string): number {
   141→  const digits = raw
   142→    .replace(/[۰-۹]/g, (d) => String("۰۱۲۳۴۵۶۷۸۹".indexOf(d)))
   143→    .replace(/[٠-٩]/g, (d) => String("٠١٢٣٤٥٦٧٨٩".indexOf(d)))
   144→    .replace(/[^0-9]/g, "");
   145→  return digits ? parseInt(digits, 10) : 0;
   146→}
   147→
   148→function formatMoneyInput(raw: string): string {
   149→  const n = parseMoney(raw);
   150→  return n ? new Intl.NumberFormat("en-US").format(n) : "";
   151→}

   153→type BracketResult = { b15: number; b20: number; b25: number; total: number };
   154→
   155→function progressiveTax(income: number, b1: number, b2: number): BracketResult {
   156→  const safe = Math.max(0, income);
   157→  const p15 = Math.min(safe, b1);
   158→  const p20 = Math.max(0, Math.min(safe, b2) - b1);
   159→  const p25 = Math.max(0, safe - b2);
   160→  const total = p15 * 0.15 + p20 * 0.2 + p25 * 0.25;
   161→  return { b15: p15 * 0.15, b20: p20 * 0.2, b25: p25 * 0.25, total };
   162→}
